using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using Microsoft.AspNetCore.Mvc.Rendering;
using OnlineStoreContracts.BindingModels;
using OnlineStoreContracts.BusinessLogicsContracts;
using OnlineStoreContracts.SearchModels;
using OnlineStoreDataModels.Enums;
using System.Security.Claims;

namespace OnlineStoreWeb.Controllers
{
    [Authorize]
    public class OrderController : Controller
    {
        private readonly IOrderLogic _orderLogic;

        public OrderController(IOrderLogic orderLogic)
        {
            _orderLogic = orderLogic;
        }

        [HttpGet]
        public IActionResult Index(int? status, int page = 1)
        {
            const int pageSize = 5;

            if (page < 1)
            {
                page = 1;
            }

            var searchModel = new OrderSearchModel
            {
                Status = status.HasValue ? (OrderStatus)status.Value : null,
                Page = page,
                PageSize = pageSize
            };

            if (!IsAdmin())
            {
                searchModel.UserId = GetCurrentUserId();
            }

            var orders = _orderLogic.ReadList(searchModel);
            var totalCount = _orderLogic.GetCount(searchModel);
            var totalPages = (int)Math.Ceiling(totalCount / (double)pageSize);

            ViewBag.Status = status;
            ViewBag.CurrentPage = page;
            ViewBag.TotalPages = totalPages;
            ViewBag.TotalCount = totalCount;
            ViewBag.IsAdmin = IsAdmin();

            return View(orders);
        }

        [HttpGet]
        public IActionResult Details(int id)
        {
            var order = _orderLogic.ReadElement(new OrderSearchModel
            {
                Id = id
            });

            if (order == null)
            {
                return NotFound();
            }

            if (!CanAccessOrder(order.UserId))
            {
                return Forbid();
            }

            return View(order);
        }

        [Authorize(Roles = "Администратор")]
        [HttpGet]
        public IActionResult Edit(int id)
        {
            var order = _orderLogic.ReadElement(new OrderSearchModel
            {
                Id = id
            });

            if (order == null)
            {
                return NotFound();
            }

            var model = new OrderBindingModel
            {
                Id = order.Id,
                UserId = order.UserId,
                CreatedDate = order.CreatedDate,
                Status = order.Status,
                TotalPrice = order.TotalPrice
            };

            ViewBag.Statuses = new SelectList(Enum.GetValues<OrderStatus>());

            return View(model);
        }

        [Authorize(Roles = "Администратор")]
        [HttpPost]
        [ValidateAntiForgeryToken]
        public IActionResult Edit(OrderBindingModel model)
        {
            if (!ModelState.IsValid)
            {
                ViewBag.Statuses = new SelectList(Enum.GetValues<OrderStatus>());
                return View(model);
            }

            try
            {
                _orderLogic.Update(model);
                TempData["Message"] = "Статус заказа успешно изменен";
                return RedirectToAction(nameof(Index));
            }
            catch (Exception ex)
            {
                ModelState.AddModelError(string.Empty, ex.Message);
                ViewBag.Statuses = new SelectList(Enum.GetValues<OrderStatus>());
                return View(model);
            }
        }

        private int GetCurrentUserId()
        {
            var userId = User.FindFirstValue(ClaimTypes.NameIdentifier);

            if (string.IsNullOrWhiteSpace(userId))
            {
                return 0;
            }

            return int.Parse(userId);
        }

        private bool IsAdmin()
        {
            return User.IsInRole(UserRole.Администратор.ToString());
        }

        private bool CanAccessOrder(int orderUserId)
        {
            return IsAdmin() || orderUserId == GetCurrentUserId();
        }
    }
}
