@model List<OrderViewModel>

@{
    ViewData["Title"] = "Заказы";

    int? status = ViewBag.Status as int?;
    bool isAdmin = ViewBag.IsAdmin ?? false;

    int currentPage = ViewBag.CurrentPage ?? 1;
    int totalPages = ViewBag.TotalPages ?? 1;
    int totalCount = ViewBag.TotalCount ?? 0;
}

<h2 class="mb-3">Заказы</h2>

@if (TempData["Message"] != null)
{
    <div class="alert alert-success">
        @TempData["Message"]
    </div>
}

<form method="get" class="row g-2 mb-3">
    <div class="col-12 col-md-4">
        <select name="status" class="form-select">
            <option value="">Все статусы</option>
            @foreach (OrderStatus item in Enum.GetValues<OrderStatus>())
            {
                var value = (int)item;

                if (status.HasValue && status.Value == value)
                {
                    <option value="@value" selected>@item</option>
                }
                else
                {
                    <option value="@value">@item</option>
                }
            }
        </select>
    </div>

    <div class="col-12 col-md-8">
        <button type="submit" class="btn btn-primary">Найти</button>
        <a asp-action="Index" class="btn btn-secondary">Сбросить</a>
    </div>
</form>

<p>Всего заказов: <strong>@totalCount</strong></p>

<div class="table-responsive">
    <table class="table table-bordered table-striped align-middle">
        <thead>
            <tr>
                <th>Номер</th>
                @if (isAdmin)
                {
                    <th>Пользователь</th>
                }
                <th>Дата</th>
                <th>Статус</th>
                <th>Сумма</th>
                <th style="width: 220px;">Действия</th>
            </tr>
        </thead>
        <tbody>
            @if (Model.Count == 0)
            {
                <tr>
                    <td colspan="@(isAdmin ? 6 : 5)" class="text-center">Заказы не найдены</td>
                </tr>
            }
            else
            {
                foreach (var order in Model)
                {
                    <tr>
                        <td>№@order.Id</td>

                        @if (isAdmin)
                        {
                            <td>@order.UserName</td>
                        }

                        <td>@order.CreatedDate.ToLocalTime().ToString("dd.MM.yyyy HH:mm")</td>
                        <td>@order.Status</td>
                        <td>@order.TotalPrice.ToString("0.00")</td>
                        <td>
                            <a asp-action="Details" asp-route-id="@order.Id" class="btn btn-sm btn-info">Детали</a>

                            @if (isAdmin)
                            {
                                <a asp-action="Edit" asp-route-id="@order.Id" class="btn btn-sm btn-warning">Статус</a>
                            }
                        </td>
                    </tr>
                }
            }
        </tbody>
    </table>
</div>

@if (totalPages > 1)
{
    <nav>
        <ul class="pagination">
            @for (int i = 1; i <= totalPages; i++)
            {
                <li class="page-item @(i == currentPage ? "active" : "")">
                    <a class="page-link"
                       asp-action="Index"
                       asp-route-page="@i"
                       asp-route-status="@status">
                        @i
                    </a>
                </li>
            }
        </ul>
    </nav>
}
