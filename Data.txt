@model List<CartItemViewModel>

@{
    ViewData["Title"] = "Корзина";
    decimal totalPrice = ViewBag.TotalPrice ?? 0;
}

<h2 class="mb-3">Корзина</h2>

@if (TempData["Message"] != null)
{
    <div class="alert alert-success">
        @TempData["Message"]
    </div>
}

@if (TempData["Error"] != null)
{
    <div class="alert alert-danger">
        @TempData["Error"]
    </div>
}

@if (Model.Count == 0)
{
    <div class="alert alert-info">
        Корзина пуста.
    </div>

    <a asp-controller="Product" asp-action="Index" class="btn btn-primary">Перейти к товарам</a>
}
else
{
    <div class="table-responsive">
        <table class="table table-bordered table-striped align-middle">
            <thead>
                <tr>
                    <th>Товар</th>
                    <th>Цена</th>
                    <th>Количество</th>
                    <th>Сумма</th>
                    <th style="width: 250px;">Действия</th>
                </tr>
            </thead>
            <tbody>
                @foreach (var item in Model)
                {
                    <tr>
                        <td>@item.ProductName</td>
                        <td>@item.ProductPrice.ToString("0.00")</td>
                        <td>
                            <form asp-action="UpdateQuantity" method="post" class="d-flex gap-2">
                                @Html.AntiForgeryToken()

                                <input type="hidden" name="id" value="@item.Id" />
                                <input type="number" name="quantity" value="@item.Quantity" min="1" class="form-control" style="max-width: 100px;" />

                                <button type="submit" class="btn btn-sm btn-warning">Обновить</button>
                            </form>
                        </td>
                        <td>@item.TotalPrice.ToString("0.00")</td>
                        <td>
                            <form asp-action="Delete" method="post" class="d-inline">
                                @Html.AntiForgeryToken()

                                <input type="hidden" name="id" value="@item.Id" />

                                <button type="submit" class="btn btn-sm btn-danger">Удалить</button>
                            </form>
                        </td>
                    </tr>
                }
            </tbody>
        </table>
    </div>

    <h4 class="mb-3">Итого: @totalPrice.ToString("0.00")</h4>

    <div class="d-flex gap-2">
        <a asp-controller="Product" asp-action="Index" class="btn btn-secondary">Продолжить покупки</a>

        <form asp-action="Checkout" method="post">
            @Html.AntiForgeryToken()
            <button type="submit" class="btn btn-success">Оформить заказ</button>
        </form>

        <form asp-action="Clear" method="post">
            @Html.AntiForgeryToken()
            <button type="submit" class="btn btn-danger">Очистить корзину</button>
        </form>
    </div>
}
