using OnlineStoreContracts.BindingModels;
using OnlineStoreContracts.BusinessLogicsContracts;
using OnlineStoreContracts.SearchModels;
using OnlineStoreContracts.StoragesContracts;
using OnlineStoreContracts.ViewModels;

namespace OnlineStoreBusinessLogic.BusinessLogics
{
    public class CartItemLogic : ICartItemLogic
    {
        private readonly ICartItemStorage _cartItemStorage;
        private readonly IProductStorage _productStorage;
        private readonly IUserStorage _userStorage;

        public CartItemLogic(ICartItemStorage cartItemStorage, IProductStorage productStorage, IUserStorage userStorage)
        {
            _cartItemStorage = cartItemStorage;
            _productStorage = productStorage;
            _userStorage = userStorage;
        }

        public List<CartItemViewModel> ReadList(CartItemSearchModel? model)
        {
            if (model == null)
            {
                return _cartItemStorage.GetFullList();
            }

            return _cartItemStorage.GetFilteredList(model);
        }

        public CartItemViewModel? ReadElement(CartItemSearchModel model)
        {
            if (model == null)
            {
                throw new ArgumentNullException(nameof(model));
            }

            return _cartItemStorage.GetElement(model);
        }

        public bool AddToCart(CartItemBindingModel model)
        {
            CheckModel(model);

            var product = _productStorage.GetElement(new ProductSearchModel
            {
                Id = model.ProductId
            });

            if (product == null)
            {
                throw new InvalidOperationException("Товар не найден");
            }

            var existingCartItem = _cartItemStorage.GetElement(new CartItemSearchModel
            {
                UserId = model.UserId,
                ProductId = model.ProductId
            });

            var newQuantity = model.Quantity;

            if (existingCartItem != null)
            {
                newQuantity += existingCartItem.Quantity;
            }

            if (product.Quantity < newQuantity)
            {
                throw new InvalidOperationException("Недостаточно товара на складе");
            }

            if (existingCartItem != null)
            {
                var updatedCartItem = new CartItemBindingModel
                {
                    Id = existingCartItem.Id,
                    UserId = existingCartItem.UserId,
                    ProductId = existingCartItem.ProductId,
                    Quantity = newQuantity
                };

                var updateResult = _cartItemStorage.Update(updatedCartItem);
                return updateResult != null;
            }

            var insertResult = _cartItemStorage.Insert(model);
            return insertResult != null;
        }

        public bool UpdateQuantity(CartItemBindingModel model)
        {
            if (model == null)
            {
                throw new ArgumentNullException(nameof(model));
            }

            if (model.Id <= 0)
            {
                throw new InvalidOperationException("Некорректный идентификатор позиции корзины");
            }

            if (model.Quantity <= 0)
            {
                throw new InvalidOperationException("Количество должно быть больше 0");
            }

            var existingCartItem = _cartItemStorage.GetElement(new CartItemSearchModel
            {
                Id = model.Id
            });

            if (existingCartItem == null)
            {
                throw new InvalidOperationException("Позиция корзины не найдена");
            }

            var product = _productStorage.GetElement(new ProductSearchModel
            {
                Id = existingCartItem.ProductId
            });

            if (product == null)
            {
                throw new InvalidOperationException("Товар не найден");
            }

            if (product.Quantity < model.Quantity)
            {
                throw new InvalidOperationException("Недостаточно товара на складе");
            }

            model.UserId = existingCartItem.UserId;
            model.ProductId = existingCartItem.ProductId;

            var result = _cartItemStorage.Update(model);
            return result != null;
        }

        public bool Delete(CartItemBindingModel model)
        {
            if (model == null)
            {
                throw new ArgumentNullException(nameof(model));
            }

            if (model.Id <= 0)
            {
                throw new InvalidOperationException("Некорректный идентификатор позиции корзины");
            }

            var result = _cartItemStorage.Delete(model);
            return result != null;
        }

        public bool ClearCart(int userId)
        {
            if (userId <= 0)
            {
                throw new InvalidOperationException("Пользователь не указан");
            }

            return _cartItemStorage.ClearCart(userId);
        }

        public decimal GetCartTotal(int userId)
        {
            if (userId <= 0)
            {
                throw new InvalidOperationException("Пользователь не указан");
            }

            var cartItems = _cartItemStorage.GetFilteredList(new CartItemSearchModel
            {
                UserId = userId
            });

            return cartItems.Sum(x => x.TotalPrice);
        }

        private void CheckModel(CartItemBindingModel model)
        {
            if (model == null)
            {
                throw new ArgumentNullException(nameof(model));
            }

            if (model.UserId <= 0)
            {
                throw new InvalidOperationException("Пользователь не указан");
            }

            if (model.ProductId <= 0)
            {
                throw new InvalidOperationException("Товар не выбран");
            }

            if (model.Quantity <= 0)
            {
                throw new InvalidOperationException("Количество должно быть больше 0");
            }

            var user = _userStorage.GetElement(new UserSearchModel
            {
                Id = model.UserId
            });

            if (user == null)
            {
                throw new InvalidOperationException("Пользователь не найден");
            }

            var product = _productStorage.GetElement(new ProductSearchModel
            {
                Id = model.ProductId
            });

            if (product == null)
            {
                throw new InvalidOperationException("Товар не найден");
            }
        }
    }
}
