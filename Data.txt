using Microsoft.EntityFrameworkCore;
using OnlineStoreContracts.BindingModels;
using OnlineStoreContracts.SearchModels;
using OnlineStoreContracts.StoragesContracts;
using OnlineStoreContracts.ViewModels;
using OnlineStoreDatabaseImplement.Database;
using OnlineStoreDatabaseImplement.Models;

namespace OnlineStoreDatabaseImplement.Storages
{
    public class ProductStorage : IProductStorage
    {
        private readonly OnlineStoreDatabase _context;

        public ProductStorage(OnlineStoreDatabase context)
        {
            _context = context;
        }

        public List<ProductViewModel> GetFullList()
        {
            return _context.Products
                .Include(x => x.Category)
                .OrderBy(x => x.Name)
                .Select(x => x.GetViewModel)
                .ToList();
        }

        public List<ProductViewModel> GetFilteredList(ProductSearchModel model)
        {
            IQueryable<Product> query = ApplyFilter(model)
                .OrderBy(x => x.Name);

            if (model.Page.HasValue && model.PageSize.HasValue)
            {
                query = query
                    .Skip((model.Page.Value - 1) * model.PageSize.Value)
                    .Take(model.PageSize.Value);
            }

            return query
                .Select(x => x.GetViewModel)
                .ToList();
        }

        public ProductViewModel? GetElement(ProductSearchModel model)
        {
            if (!model.Id.HasValue)
            {
                return null;
            }

            return _context.Products
                .Include(x => x.Category)
                .FirstOrDefault(x => x.Id == model.Id.Value)
                ?.GetViewModel;
        }

        public int GetCount(ProductSearchModel? model)
        {
            if (model == null)
            {
                return _context.Products.Count();
            }

            return ApplyFilter(model).Count();
        }

        public ProductViewModel? Insert(ProductBindingModel model)
        {
            var product = Product.Create(model);
            _context.Products.Add(product);
            _context.SaveChanges();

            return GetElement(new ProductSearchModel
            {
                Id = product.Id
            });
        }

        public ProductViewModel? Update(ProductBindingModel model)
        {
            var product = _context.Products.FirstOrDefault(x => x.Id == model.Id);

            if (product == null)
            {
                return null;
            }

            product.Update(model);
            _context.SaveChanges();

            return GetElement(new ProductSearchModel
            {
                Id = product.Id
            });
        }

        public ProductViewModel? Delete(ProductBindingModel model)
        {
            var product = _context.Products.FirstOrDefault(x => x.Id == model.Id);

            if (product == null)
            {
                return null;
            }

            _context.Products.Remove(product);
            _context.SaveChanges();

            return product.GetViewModel;
        }

        public bool DecreaseQuantity(int productId, int quantity)
        {
            var product = _context.Products.FirstOrDefault(x => x.Id == productId);

            if (product == null)
            {
                return false;
            }

            if (product.Quantity < quantity)
            {
                return false;
            }

            product.Quantity -= quantity;
            _context.SaveChanges();

            return true;
        }

        private IQueryable<Product> ApplyFilter(ProductSearchModel model)
        {
            var query = _context.Products
                .Include(x => x.Category)
                .AsQueryable();

            if (model.Id.HasValue)
            {
                query = query.Where(x => x.Id == model.Id.Value);
            }

            if (model.CategoryId.HasValue)
            {
                query = query.Where(x => x.CategoryId == model.CategoryId.Value);
            }

            if (!string.IsNullOrWhiteSpace(model.SearchText))
            {
                var searchText = model.SearchText.ToLower();

                query = query.Where(x =>
                    x.Name.ToLower().Contains(searchText) ||
                    (x.Description != null && x.Description.ToLower().Contains(searchText)));
            }

            if (model.MinPrice.HasValue)
            {
                query = query.Where(x => x.Price >= model.MinPrice.Value);
            }

            if (model.MaxPrice.HasValue)
            {
                query = query.Where(x => x.Price <= model.MaxPrice.Value);
            }

            return query;
        }
    }
}
