using OnlineStoreContracts.BindingModels;
using OnlineStoreContracts.BusinessLogicsContracts;
using OnlineStoreContracts.SearchModels;
using OnlineStoreContracts.StoragesContracts;
using OnlineStoreContracts.ViewModels;

namespace OnlineStoreBusinessLogic.BusinessLogics
{
    public class OrderLogic : IOrderLogic
    {
        private readonly IOrderStorage _orderStorage;
        private readonly IUserStorage _userStorage;
        private readonly ICartItemStorage _cartItemStorage;

        public OrderLogic(IOrderStorage orderStorage, IUserStorage userStorage, ICartItemStorage cartItemStorage)
        {
            _orderStorage = orderStorage;
            _userStorage = userStorage;
            _cartItemStorage = cartItemStorage;
        }

        public List<OrderViewModel> ReadList(OrderSearchModel? model)
        {
            if (model == null)
            {
                return _orderStorage.GetFullList();
            }

            return _orderStorage.GetFilteredList(model);
        }

        public OrderViewModel? ReadElement(OrderSearchModel model)
        {
            if (model == null)
            {
                throw new ArgumentNullException(nameof(model));
            }

            return _orderStorage.GetElement(model);
        }

        public int GetCount(OrderSearchModel? model)
        {
            return _orderStorage.GetCount(model);
        }

        public bool Create(OrderBindingModel model)
        {
            CheckModel(model);

            var result = _orderStorage.Insert(model);
            return result != null;
        }

        public bool Update(OrderBindingModel model)
        {
            if (model == null)
            {
                throw new ArgumentNullException(nameof(model));
            }

            if (model.Id <= 0)
            {
                throw new InvalidOperationException("Некорректный идентификатор заказа");
            }

            var existingOrder = _orderStorage.GetElement(new OrderSearchModel
            {
                Id = model.Id
            });

            if (existingOrder == null)
            {
                throw new InvalidOperationException("Заказ не найден");
            }

            model.UserId = existingOrder.UserId;
            model.CreatedDate = existingOrder.CreatedDate;
            model.TotalPrice = existingOrder.TotalPrice;

            var result = _orderStorage.Update(model);
            return result != null;
        }

        public bool Delete(OrderBindingModel model)
        {
            if (model == null)
            {
                throw new ArgumentNullException(nameof(model));
            }

            if (model.Id <= 0)
            {
                throw new InvalidOperationException("Некорректный идентификатор заказа");
            }

            var result = _orderStorage.Delete(model);
            return result != null;
        }

        public bool Checkout(int userId)
        {
            if (userId <= 0)
            {
                throw new InvalidOperationException("Пользователь не указан");
            }

            var user = _userStorage.GetElement(new UserSearchModel
            {
                Id = userId
            });

            if (user == null)
            {
                throw new InvalidOperationException("Пользователь не найден");
            }

            var cartItems = _cartItemStorage.GetFilteredList(new CartItemSearchModel
            {
                UserId = userId
            });

            if (cartItems.Count == 0)
            {
                throw new InvalidOperationException("Корзина пуста");
            }

            return _orderStorage.Checkout(userId);
        }

        private void CheckModel(OrderBindingModel model)
        {
            if (model == null)
            {
                throw new ArgumentNullException(nameof(model));
            }

            if (model.UserId <= 0)
            {
                throw new InvalidOperationException("Пользователь заказа не указан");
            }

            var user = _userStorage.GetElement(new UserSearchModel
            {
                Id = model.UserId
            });

            if (user == null)
            {
                throw new InvalidOperationException("Пользователь не найден");
            }

            if (model.TotalPrice < 0)
            {
                throw new InvalidOperationException("Сумма заказа не может быть отрицательной");
            }
        }
    }
}
