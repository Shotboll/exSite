using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using OnlineStoreContracts.BindingModels;
using OnlineStoreContracts.BusinessLogicsContracts;
using OnlineStoreContracts.SearchModels;

namespace OnlineStoreWeb.Controllers
{
    [Authorize(Roles = "Администратор")]
    public class CategoryController : Controller
    {
        private readonly ICategoryLogic _categoryLogic;

        public CategoryController(ICategoryLogic categoryLogic)
        {
            _categoryLogic = categoryLogic;
        }

        [HttpGet]
        public IActionResult Index(string? searchText)
        {
            var categories = _categoryLogic.ReadList(new CategorySearchModel
            {
                SearchText = searchText
            });

            ViewBag.SearchText = searchText;

            return View(categories);
        }

        [HttpGet]
        public IActionResult Create()
        {
            return View(new CategoryBindingModel());
        }

        [HttpPost]
        [ValidateAntiForgeryToken]
        public IActionResult Create(CategoryBindingModel model)
        {
            if (!ModelState.IsValid)
            {
                return View(model);
            }

            try
            {
                _categoryLogic.Create(model);
                TempData["Message"] = "Категория успешно добавлена";
                return RedirectToAction(nameof(Index));
            }
            catch (Exception ex)
            {
                ModelState.AddModelError(string.Empty, ex.Message);
                return View(model);
            }
        }

        [HttpGet]
        public IActionResult Edit(int id)
        {
            var category = _categoryLogic.ReadElement(new CategorySearchModel
            {
                Id = id
            });

            if (category == null)
            {
                return NotFound();
            }

            var model = new CategoryBindingModel
            {
                Id = category.Id,
                Name = category.Name,
                Description = category.Description
            };

            return View(model);
        }

        [HttpPost]
        [ValidateAntiForgeryToken]
        public IActionResult Edit(CategoryBindingModel model)
        {
            if (!ModelState.IsValid)
            {
                return View(model);
            }

            try
            {
                _categoryLogic.Update(model);
                TempData["Message"] = "Категория успешно изменена";
                return RedirectToAction(nameof(Index));
            }
            catch (Exception ex)
            {
                ModelState.AddModelError(string.Empty, ex.Message);
                return View(model);
            }
        }

        [HttpGet]
        public IActionResult Delete(int id)
        {
            var category = _categoryLogic.ReadElement(new CategorySearchModel
            {
                Id = id
            });

            if (category == null)
            {
                return NotFound();
            }

            var model = new CategoryBindingModel
            {
                Id = category.Id,
                Name = category.Name,
                Description = category.Description
            };

            return View(model);
        }

        [HttpPost]
        [ValidateAntiForgeryToken]
        public IActionResult Delete(CategoryBindingModel model)
        {
            try
            {
                _categoryLogic.Delete(model);
                TempData["Message"] = "Категория успешно удалена";
                return RedirectToAction(nameof(Index));
            }
            catch
            {
                ModelState.AddModelError(string.Empty, "Невозможно удалить категорию, потому что она используется в товарах");
                return View(model);
            }
        }
    }
}
