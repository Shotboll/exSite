using Microsoft.EntityFrameworkCore;
using OnlineStoreContracts.BindingModels;
using OnlineStoreContracts.SearchModels;
using OnlineStoreContracts.StoragesContracts;
using OnlineStoreContracts.ViewModels;
using OnlineStoreDatabaseImplement.Database;
using OnlineStoreDatabaseImplement.Models;

namespace OnlineStoreDatabaseImplement.Storages
{
    public class CartItemStorage : ICartItemStorage
    {
        private readonly OnlineStoreDatabase _context;

        public CartItemStorage(OnlineStoreDatabase context)
        {
            _context = context;
        }

        public List<CartItemViewModel> GetFullList()
        {
            return _context.CartItems
                .Include(x => x.User)
                .Include(x => x.Product)
                .OrderBy(x => x.Id)
                .Select(x => x.GetViewModel)
                .ToList();
        }

        public List<CartItemViewModel> GetFilteredList(CartItemSearchModel model)
        {
            var query = _context.CartItems
                .Include(x => x.User)
                .Include(x => x.Product)
                .AsQueryable();

            if (model.Id.HasValue)
            {
                query = query.Where(x => x.Id == model.Id.Value);
            }

            if (model.UserId.HasValue)
            {
                query = query.Where(x => x.UserId == model.UserId.Value);
            }

            if (model.ProductId.HasValue)
            {
                query = query.Where(x => x.ProductId == model.ProductId.Value);
            }

            return query
                .OrderBy(x => x.Id)
                .Select(x => x.GetViewModel)
                .ToList();
        }

        public CartItemViewModel? GetElement(CartItemSearchModel model)
        {
            var query = _context.CartItems
                .Include(x => x.User)
                .Include(x => x.Product)
                .AsQueryable();

            if (model.Id.HasValue)
            {
                return query
                    .FirstOrDefault(x => x.Id == model.Id.Value)
                    ?.GetViewModel;
            }

            if (model.UserId.HasValue && model.ProductId.HasValue)
            {
                return query
                    .FirstOrDefault(x => x.UserId == model.UserId.Value && x.ProductId == model.ProductId.Value)
                    ?.GetViewModel;
            }

            return null;
        }

        public CartItemViewModel? Insert(CartItemBindingModel model)
        {
            var cartItem = CartItem.Create(model);
            _context.CartItems.Add(cartItem);
            _context.SaveChanges();

            return GetElement(new CartItemSearchModel
            {
                Id = cartItem.Id
            });
        }

        public CartItemViewModel? Update(CartItemBindingModel model)
        {
            var cartItem = _context.CartItems.FirstOrDefault(x => x.Id == model.Id);

            if (cartItem == null)
            {
                return null;
            }

            cartItem.Update(model);
            _context.SaveChanges();

            return GetElement(new CartItemSearchModel
            {
                Id = cartItem.Id
            });
        }

        public CartItemViewModel? Delete(CartItemBindingModel model)
        {
            var cartItem = _context.CartItems.FirstOrDefault(x => x.Id == model.Id);

            if (cartItem == null)
            {
                return null;
            }

            _context.CartItems.Remove(cartItem);
            _context.SaveChanges();

            return cartItem.GetViewModel;
        }

        public bool ClearCart(int userId)
        {
            var cartItems = _context.CartItems
                .Where(x => x.UserId == userId)
                .ToList();

            _context.CartItems.RemoveRange(cartItems);
            _context.SaveChanges();

            return true;
        }
    }
}
