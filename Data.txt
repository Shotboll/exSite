@model List<ProductViewModel>

@{
    ViewData["Title"] = "Товары";

    string? searchText = ViewBag.SearchText as string;
    int? categoryId = ViewBag.CategoryId as int?;
    decimal? minPrice = ViewBag.MinPrice as decimal?;
    decimal? maxPrice = ViewBag.MaxPrice as decimal?;

    int currentPage = ViewBag.CurrentPage ?? 1;
    int totalPages = ViewBag.TotalPages ?? 1;
    int totalCount = ViewBag.TotalCount ?? 0;
}

<h2 class="mb-3">Товары</h2>

@if (TempData["Message"] != null)
{
    <div class="alert alert-success">
        @TempData["Message"]
    </div>
}

@if (TempData["Error"] != null)
{
    <div class="alert alert-danger">
        @TempData["Error"]
    </div>
}

<form method="get" class="row g-2 mb-3">
    <div class="col-12 col-md-3">
        <input type="text" name="searchText" value="@searchText" class="form-control" placeholder="Поиск товара" />
    </div>

    <div class="col-12 col-md-3">
        <select name="categoryId" class="form-select">
            <option value="">Все категории</option>
            @foreach (var item in ViewBag.Categories)
            {
                if (categoryId.HasValue && categoryId.Value.ToString() == item.Value)
                {
                    <option value="@item.Value" selected>@item.Text</option>
                }
                else
                {
                    <option value="@item.Value">@item.Text</option>
                }
            }
        </select>
    </div>

    <div class="col-6 col-md-2">
        <input type="number" step="0.01" name="minPrice" value="@minPrice" class="form-control" placeholder="Цена от" />
    </div>

    <div class="col-6 col-md-2">
        <input type="number" step="0.01" name="maxPrice" value="@maxPrice" class="form-control" placeholder="Цена до" />
    </div>

    <div class="col-12 col-md-2">
        <button type="submit" class="btn btn-primary">Найти</button>
        <a asp-action="Index" class="btn btn-secondary">Сбросить</a>
    </div>
</form>

@if (User.IsInRole("Администратор"))
{
    <p>
        <a asp-action="Create" class="btn btn-success">Добавить товар</a>
    </p>
}

<p>Всего товаров: <strong>@totalCount</strong></p>

<div class="table-responsive">
    <table class="table table-bordered table-striped align-middle">
        <thead>
            <tr>
                <th>Название</th>
                <th>Категория</th>
                <th>Цена</th>
                <th>Остаток</th>
                <th style="width: 330px;">Действия</th>
            </tr>
        </thead>
        <tbody>
            @if (Model.Count == 0)
            {
                <tr>
                    <td colspan="5" class="text-center">Товары не найдены</td>
                </tr>
            }
            else
            {
                foreach (var product in Model)
                {
                    <tr>
                        <td>@product.Name</td>
                        <td>@product.CategoryName</td>
                        <td>@product.Price.ToString("0.00")</td>
                        <td>@product.Quantity</td>
                        <td>
                            <a asp-action="Details" asp-route-id="@product.Id" class="btn btn-sm btn-info">Детали</a>

                            @if (User.Identity != null && User.Identity.IsAuthenticated && !User.IsInRole("Администратор"))
                            {
                                <form asp-action="AddToCart" method="post" class="d-inline">
                                    @Html.AntiForgeryToken()
                                    <input type="hidden" name="productId" value="@product.Id" />
                                    <input type="hidden" name="quantity" value="1" />
                                    <button type="submit" class="btn btn-sm btn-success" @(product.Quantity <= 0 ? "disabled" : "")>
                                        В корзину
                                    </button>
                                </form>
                            }

                            @if (User.IsInRole("Администратор"))
                            {
                                <a asp-action="Edit" asp-route-id="@product.Id" class="btn btn-sm btn-warning">Изменить</a>
                                <a asp-action="Delete" asp-route-id="@product.Id" class="btn btn-sm btn-danger">Удалить</a>
                            }
                        </td>
                    </tr>
                }
            }
        </tbody>
    </table>
</div>

@if (totalPages > 1)
{
    <nav>
        <ul class="pagination">
            @for (int i = 1; i <= totalPages; i++)
            {
                <li class="page-item @(i == currentPage ? "active" : "")">
                    <a class="page-link"
                       asp-action="Index"
                       asp-route-page="@i"
                       asp-route-searchText="@searchText"
                       asp-route-categoryId="@categoryId"
                       asp-route-minPrice="@minPrice"
                       asp-route-maxPrice="@maxPrice">
                        @i
                    </a>
                </li>
            }
        </ul>
    </nav>
}
